cmake_minimum_required(VERSION 3.21)
project(g923_mac_bridge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_subdirectory(vendor/wheelio)
add_executable(tf-packet-test tests/tf_packet_test.cpp)
target_include_directories(tf-packet-test PRIVATE src)
add_test(NAME tf-packets COMMAND tf-packet-test)
if(APPLE)
  file(GLOB CORE vendor/wheelio/src/*.cpp)
  add_executable(g923-ffb src/ffb_main.cpp ${CORE}
    vendor/wheelio/bridge/macos/bridge_server.cpp
    vendor/wheelio/bridge/common/ffb_mapping.cpp vendor/wheelio/bridge/common/dinput_effects.cpp)
  target_include_directories(g923-ffb PRIVATE vendor/wheelio/include vendor/wheelio/bridge/include)
  target_link_libraries(g923-ffb PRIVATE "-framework IOKit" "-framework CoreFoundation")
  add_executable(g923-input src/input_bridge_main.cpp)
  target_link_libraries(g923-input PRIVATE "-framework IOKit" "-framework CoreFoundation")
  add_executable(g923-hid src/hid_main.cpp)
  target_link_libraries(g923-hid PRIVATE "-framework IOKit" "-framework CoreFoundation")
  add_executable(g923-tf src/tf_main.cpp)
  target_link_libraries(g923-tf PRIVATE "-framework IOKit" "-framework CoreFoundation")
endif()
if(WIN32)
  add_executable(dinput-probe src/dinput_probe.cpp)
  target_link_libraries(dinput-probe PRIVATE dinput8 dxguid)
  target_link_options(dinput-probe PRIVATE -static-libgcc -static-libstdc++)
endif()

# Hardware-free failure-path checks for restoring the original SDK force mode.
add_executable(test-zero-mode tests/test_zero_mode.cpp)
add_test(NAME sdk-zero-mode COMMAND test-zero-mode)
add_executable(test-tf-plan tests/test_tf_plan.cpp)
add_test(NAME sdk-tf-test-plan COMMAND test-tf-plan)

add_executable(test-combined-plan tests/test_combined_plan.cpp)
add_test(NAME sdk-combined-plan COMMAND test-combined-plan)

# Keep assertion-based offline checks active in optimized test builds.
foreach(g923_test_target tf-packet-test test-zero-mode test-tf-plan test-combined-plan)
  if(MSVC)
    target_compile_options(${g923_test_target} PRIVATE /UNDEBUG)
  else()
    target_compile_options(${g923_test_target} PRIVATE -UNDEBUG)
  endif()
endforeach()
